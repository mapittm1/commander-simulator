/** Membership mutation and exposed views must not hide live native card traits. */
public class CommanderKeywordMembershipTest extends AITest {
    private Player actor(Game game) { return game.getPlayers().getFirst(); }
    private Game setup(int depth) {
        List<RegisteredPlayer> seats = new ArrayList<>();
        for (int i = 0; i < 4; i++) seats.add(RegisteredPlayer.forCommander(new Deck()).setPlayer(new LobbyPlayerAi("trait-gather-" + i, null)));
        GameRules rules = new GameRules(GameType.Commander);
        Game game = new Game(seats, rules, new Match(rules, seats, "Live trait gathering"));
        game.setNoGUIUser(); game.setAge(GameStage.Play);
        for (Player player : game.getPlayers()) {
            player.setLife(100, null);
            for (int i = 0; i < 20; i++) addCardToZone("Forest", player, ZoneType.Library);
        }
        addCardToZone("Forest", actor(game), ZoneType.Battlefield);
        game.getPhaseHandler().devModeSet(PhaseType.MAIN1, actor(game), 5);
        game.getAction().checkStaticAbilities(false); game.getTriggerHandler().resetActiveTriggers(); game.copyLastState();
        for (int i = 0; i < depth; i++) {
            GameCopier copier = new GameCopier(game).withStack(true);
            game = copier.makeCopy(null, actor(game)); copier.release();
        }
        return game;
    }

    private void observe(Card card, boolean expected) {
        assertEquals(card.getStaticAbilities().stream().anyMatch(st -> st.checkMode(StaticAbilityMode.CantBeActivated)), expected);
        assertEquals(card.getReplacementEffects().stream().anyMatch(re -> re.getMode() == ReplacementType.Tap), expected);
        assertEquals(card.canTap(), !expected, "Native tap replacement visibility");
    }
    private void check(String mode, int depth) {
        Game game = setup(depth); Card card = actor(game).getCardsIn(ZoneType.Battlefield).getFirst();
        KeywordCollection keywords = new KeywordCollection(); KeywordInterface harmless = keywords.add("Flying");
        KeywordInterface positive = Keyword.getInstance("Haste");
        StaticAbility ban = new StaticAbility("Mode$ CantBeActivated | AffectedZone$ Battlefield | ValidCard$ Card.Self | ValidSA$ Activated", card, card.getCurrentState());
        ReplacementEffect replacement = ReplacementHandler.parseReplacement("Event$ Tap | ActiveZones$ Battlefield | ValidCard$ Card.Self | Layer$ CantHappen", card, true);
        positive.getStaticAbilities().add(ban); positive.getReplacements().add(replacement);
        card.getCurrentState().setCachedKeywords(keywords); observe(card, false);
        if (mode.equals("keyAddRetained") || mode.equals("sameSizeView")) {
            Collection<KeywordInterface> retained = keywords.getValues(Keyword.HASTE);
            observe(card, false);
            if (mode.equals("sameSizeView")) assertTrue(keywords.getValues(Keyword.FLYING).remove(harmless));
            assertTrue(retained.add(positive)); observe(card, true); assertTrue(retained.remove(positive)); observe(card, false);
            retained.add(positive); observe(card, true); return;
        }
        if (mode.equals("addAll")) {
            keywords.addAll(List.of("Haste")); positive = keywords.getValues(Keyword.HASTE).iterator().next();
            positive.getStaticAbilities().add(ban); positive.getReplacements().add(replacement);
        } else if (mode.equals("insertAll")) keywords.insertAll(List.of(positive));
        else keywords.insert(positive);
        observe(card, true);
        if (mode.equals("copyIndependent")) {
            KeywordCollection copied = keywords.copy(card, true); card.getCurrentState().setCachedKeywords(copied);
            observe(card, true); keywords.clear(); observe(card, true); copied.removeAll(Keyword.HASTE); observe(card, false);
            keywords.insert(positive); observe(card, false); return;
        }
        if (mode.equals("traitListsLive")) {
            positive.getStaticAbilities().clear(); positive.getReplacements().clear(); observe(card, false);
            positive.getStaticAbilities().add(ban); positive.getReplacements().add(replacement); observe(card, true); return;
        }
        if (mode.equals("insert") || mode.equals("insertAll") || mode.equals("addAll") || mode.equals("removeInstance")) keywords.remove(positive);
        else if (mode.equals("removePrefix")) keywords.remove("Haste");
        else if (mode.equals("removeEnum")) keywords.removeAll(Keyword.HASTE);
        else if (mode.equals("removeInstances")) keywords.removeInstances(List.of(positive));
        else if (mode.equals("clear")) keywords.clear();
        else if (mode.equals("iteratorRemove")) {
            Iterator<KeywordInterface> iterator = keywords.iterator();
            while (iterator.hasNext()) if (iterator.next() == positive) iterator.remove();
        } else if (mode.equals("valuesRemove")) keywords.getValues().remove(positive);
        else if (mode.equals("valuesClear")) keywords.getValues().clear();
        else if (mode.equals("keyClearRetained")) {
            Collection<KeywordInterface> retained = keywords.getValues(Keyword.HASTE); retained.clear(); observe(card, false);
            retained.add(positive); observe(card, true); return;
        } else throw new IllegalArgumentException(mode);
        observe(card, false); keywords.insert(positive); observe(card, true);
    }
