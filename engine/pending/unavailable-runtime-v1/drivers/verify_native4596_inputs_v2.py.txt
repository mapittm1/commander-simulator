"""Compare fresh native test/card inputs with the completed source validation."""
from pathlib import Path
import json,hashlib,sys
ROOT=Path(__file__).resolve().parents[4];sys.path.insert(0,str(ROOT));WORK=Path(__file__).resolve().parent
load=lambda p:json.loads(p.read_bytes())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
CANDIDATE=WORK/'no-legal-attack-candidate-v2';SOURCE=WORK/'source3396-v1'
assert load(SOURCE/'evidence.json')['counts']==dict(total=3396,passed=3396,failed=0,skipped=0)
audit=load(CANDIDATE/'build-audit.json');assert audit['candidate']['native_tests']==dict(total=4596,passed=4596,failed=0,skipped=0)
build=ROOT/'data/engine-builds'/audit['build_id'];frozen=load(CANDIDATE/'source-frozen-inputs.json')
assert all(sha(ROOT/n)==h for n,h in frozen.items());driver_sha=sha(Path(__file__))
summary={}
for kind in ('test','card'):
 inputs=load(SOURCE/(kind+'-inputs.json'));line_endings=[]
 for relative,digest in inputs.items():
  original=SOURCE/'input-snapshots'/relative;fresh=build/'source'/relative
  assert sha(original)==digest,relative
  old=original.read_bytes();new=fresh.read_bytes()
  assert old==new or old.replace(b'\r\n',b'\n')==new.replace(b'\r\n',b'\n'),relative
  if old!=new:line_endings.append(relative)
 summary[kind]={'files':len(inputs),'all_inputs_match':True,'line_ending_paths':line_endings,'manifest_sha256':sha(SOURCE/(kind+'-inputs.json'))}
assert sha(Path(__file__))==driver_sha and all(sha(ROOT/n)==h for n,h in frozen.items())
record={'build_id':build.name,'source_case_count':3396,'native_case_count':4596,'inputs':summary,'driver_sha256':driver_sha,'scope':'Exact file content comparison, allowing CRLF/LF only; complements case/parameter and production-byte audits. Not full card/rules correctness proof.'}
out=CANDIDATE/'test-card-input-audit-v2.json';assert not out.exists();out.write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps({'build':build.name,'test_files':summary['test']['files'],'card_files':summary['card']['files'],'all_match':True}),flush=True)
