r=Path.cwd();sys.path.insert(0,str(r))

  trace=event.getStackTrace()
  if trace is None:continue
  names=[str(f.getMethod().getType().getName())+'.'+str(f.getMethod().getName()) for f in trace.getFrames() if f.getMethod() is not None]
  positions=[i for i,n in enumerate(names) if n.startswith('forge.card.CardDb.')]
  if not positions:continue
  end=max(positions);suffix=names[end+1:]

  stacks[' <- '.join(names[max(0,end-1):end+17])]+=1
