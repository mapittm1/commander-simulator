package forge.ai;

import forge.ai.simulation.GameCopier;
import forge.card.CardType;
import forge.card.CardTypeView;
import forge.game.Game;
import forge.game.ability.AbilityKey;
import forge.game.card.Card;
import forge.game.card.CardState;
import forge.game.card.CardTraitChanges;
import forge.game.phase.PhaseType;
import forge.game.player.Player;
import forge.game.replacement.ReplacementEffect;
import forge.game.replacement.ReplacementHandler;
import forge.game.replacement.ReplacementResult;
import forge.game.replacement.ReplacementType;
import forge.game.zone.ZoneType;
import forge.util.collect.FCollection;
import forge.util.collect.FCollectionView;
import java.lang.reflect.Field;
import java.lang.reflect.Proxy;
import java.util.ArrayList;
import java.util.List;
import org.testng.annotations.AfterMethod;
import org.testng.annotations.Test;
import static org.testng.Assert.*;

/** Supplied-board phase replacement controls, not complete legal turn histories. */
public class CommanderPhasePrecheckTest extends AITest {
    private final List<GameCopier> copies = new ArrayList<>();
    private static final class Scene {
        Game game; Player actor; Player other; Card plain;
    }
    private Scene scene() {
        Scene s = new Scene(); s.game = CommanderPayloadGraphTest.newBoard();
        s.actor = s.game.getPlayers().getFirst(); s.other = s.game.getPlayers().get(1);
        s.plain = addCardToZone("Grizzly Bears", s.actor, ZoneType.Battlefield);
        s.game.getAction().checkStaticAbilities(false);
        s.game.getTriggerHandler().resetActiveTriggers(); s.game.copyLastState(); return s;
    }
    private ReplacementResult phase(Game game, Player player, PhaseType phase) {
        var parameters = AbilityKey.mapFromAffected(player);
        parameters.put(AbilityKey.Phase, phase);
        return game.getReplacementHandler().run(ReplacementType.BeginPhase, parameters);
    }
    private ReplacementEffect skip(Card host) {
        return ReplacementHandler.parseReplacement("Event$ BeginPhase | ActiveZones$ Battlefield | Phase$ Upkeep | Skip$ True", host, true);
    }
    private CardTraitChanges traits(ReplacementEffect effect) {
        return new CardTraitChanges(List.of(), List.of(), List.of(effect), List.of(), null);
    }
    @AfterMethod public void releaseCopies() {
        for (GameCopier copier : copies) copier.release(); copies.clear();
    }
    @Test public void ordinaryUpkeepIsNotSkipped() {
        Scene s = scene(); assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.NotReplaced);
    }
    @Test public void eonHubSkipsBothPlayersUpkeeps() {
        Scene s = scene(); Card hub = addCardToZone("Eon Hub", s.actor, ZoneType.Battlefield);
        assertTrue(hub.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase));
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
        assertEquals(phase(s.game, s.other, PhaseType.UPKEEP), ReplacementResult.Skipped);
    }
    @Test public void eonHubDoesNotSkipDraw() {
        Scene s = scene(); addCardToZone("Eon Hub", s.actor, ZoneType.Battlefield);
        assertEquals(phase(s.game, s.actor, PhaseType.DRAW), ReplacementResult.NotReplaced);
    }
    @Test public void eonHubInHandDoesNotSkipUpkeep() {
        Scene s = scene(); addCardToZone("Eon Hub", s.actor, ZoneType.Hand);
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.NotReplaced);
    }
    @Test public void solitaryConfinementSkipsControllersDraw() {
        Scene s = scene(); addCardToZone("Solitary Confinement", s.actor, ZoneType.Battlefield);
        assertEquals(phase(s.game, s.actor, PhaseType.DRAW), ReplacementResult.Skipped);
    }
    @Test public void solitaryConfinementDoesNotSkipOpponentsDraw() {
        Scene s = scene(); addCardToZone("Solitary Confinement", s.actor, ZoneType.Battlefield);
        assertEquals(phase(s.game, s.other, PhaseType.DRAW), ReplacementResult.NotReplaced);
    }
    @Test public void replacementAddedAfterNegativeProbeIsVisible() {
        Scene s = scene(); s.plain.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase);
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.NotReplaced);
        s.plain.addChangedCardTraits(traits(skip(s.plain)), s.game.getNextTimestamp(), 0L, false);
        assertTrue(s.plain.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase));
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
    }
    @Test public void replacementRemovedAfterPositiveProbeIsNotReused() {
        Scene s = scene(); long timestamp = s.game.getNextTimestamp();
        s.plain.addChangedCardTraits(traits(skip(s.plain)), timestamp, 0L, false);
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
        assertTrue(s.plain.removeChangedCardTraits(timestamp, 0L));
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.NotReplaced);
    }
    @Test public void textChangedReplacementIsNotSkipped() {
        Scene s = scene(); s.plain.getChangedCardTraitsByText().put(s.game.getNextTimestamp(), 0L, traits(skip(s.plain)));
        assertTrue(s.plain.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase));
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
    }
    private void copiedHub(int depth) {
        Scene s = scene(); Card hub = addCardToZone("Eon Hub", s.actor, ZoneType.Battlefield); s.game.copyLastState();
        Game game = s.game; Player actor = s.actor; Card mapped = hub;
        for (int index = 0; index < depth; index++) {
            GameCopier copier = new GameCopier(game).withStack(true); copies.add(copier);
            game = copier.makeCopy(null, actor); actor = copier.find(actor); mapped = copier.find(mapped);
        }
        assertNotSame(mapped, hub); assertEquals(mapped.getId(), hub.getId());
        assertTrue(mapped.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase));
        assertEquals(phase(game, actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
        assertTrue(hub.isInZone(ZoneType.Battlefield));
    }
    @Test public void copiedEonHubRetainsReplacement() { copiedHub(1); }
    @Test public void repeatedCopyRetainsPhaseReplacement() { copiedHub(2); }
    private static final class CallbackCard extends Card {
        ReplacementEffect extra; int reads;
        CallbackCard(Game game, Player owner) {
            super(game.nextCardId(), null, game); setOwner(owner); setName("Phase callback contract");
            setType(CardType.parse("Artifact", false));
        }
        @Override public FCollectionView<ReplacementEffect> getReplacementEffects() {
            reads++; FCollection<ReplacementEffect> result = new FCollection<>(super.getReplacementEffects());
            if (extra != null) result.add(extra); return result;
        }
    }
    @Test public void cardSubclassReplacementCallbackStillRuns() {
        Scene s = scene(); CallbackCard host = new CallbackCard(s.game, s.actor);
        s.actor.getZone(ZoneType.Battlefield).add(host); host.extra = skip(host); int before = host.reads;
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.Skipped);
        assertTrue(host.reads > before);
    }
    @Test public void unknownTypeImplementationRetainsGathering() throws Exception {
        Scene s = scene(); CardType type = CardType.parse("Creature", false);
        CardTypeView view = (CardTypeView) Proxy.newProxyInstance(CardTypeView.class.getClassLoader(), new Class<?>[]{CardTypeView.class}, (proxy, method, arguments) -> method.invoke(type, arguments));
        Field field = CardState.class.getDeclaredField("changedType"); field.setAccessible(true); field.set(s.plain.getCurrentState(), view);
        assertTrue(s.plain.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase));
        assertEquals(phase(s.game, s.actor, PhaseType.UPKEEP), ReplacementResult.NotReplaced);
    }
    @Test public void otherEventModesAndNullRemainConservative() {
        Scene s = scene(); CardState state = s.plain.getCurrentState();
        for (ReplacementType event : ReplacementType.values()) {
            if (event != ReplacementType.Tap && event != ReplacementType.ProduceMana && event != ReplacementType.BeginPhase) assertTrue(state.mayHaveReplacementEffects(event), event.toString());
        }
        assertTrue(state.mayHaveReplacementEffects(null));
    }
    public boolean[] plainAdmissionObservation() {
        Scene s = scene(); return new boolean[]{s.plain.getCurrentState().mayHaveReplacementEffects(ReplacementType.BeginPhase)};
    }
}
