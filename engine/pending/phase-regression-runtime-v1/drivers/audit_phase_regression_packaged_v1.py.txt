"""Audit the registered phase group's packaged receipts and native case identities."""
from collections import Counter
from pathlib import Path
import json,re,sys,xml.etree.ElementTree as ET
ROOT=Path.cwd();sys.path.insert(0,str(ROOT))
from commander.engine.forge_config import file_hash,read_settings
from tests.support.draw_ending_partitions import PARTITIONS,EXPECTED_COUNTS
from tests.support.draw_ending_worker import NATIVE_CLASSES
from tests.support.native_receipts import read_test_receipts
WORK=ROOT/'data/development/commander-multiplayer-attack-research/waterbend-filter-research'
RUN=WORK/'phase-regression-packaged-v1';CANDIDATE=WORK/'phase-regression-candidate-v1'
load=lambda p:json.loads(p.read_bytes())
def cases(p,classes=None):
 result=Counter()
 for cls in ET.parse(p).findall('.//class'):
  if classes is not None and cls.get('name') not in classes:continue
  for method in cls.findall('test-method'):
   assert method.get('status')=='PASS',(cls.get('name'),method.attrib)
   if method.get('is-config')!='true':
    result[(cls.get('name')+'.'+method.get('name'),tuple((v.text or '').strip() for v in method.findall('./params/param/value')))]+=1
 return result
frozen=load(RUN/'source-frozen-inputs.json');assert all(file_hash(ROOT/n)==h for n,h in frozen.items())
outer=load(RUN/'process.json');process=outer['process']
assert process['status']=='finished' and process['returncode']==0 and process['elapsed_seconds']<=420
assert outer['configuration_restored'] and file_hash(RUN/'output.log')==outer['log_sha256']
outcome=load(RUN/'outcome.json');assert outcome['successful'] and outcome['tests_run']==1 and not outcome['skipped']
build=ROOT/load(CANDIDATE/'candidate-selection.json')['build'];settings=read_settings(build/'engine-config.json')
assert settings.verify()['native_tests']==dict(total=4932,passed=4932,failed=0,skipped=0)
ids=re.findall(r'"draw_ending_partition": "([^"\n]+)", "draw_ending_run": "([a-f0-9]{32})"',(RUN/'output.log').read_text(encoding='utf-8'))
assert len(ids)==1 and ids[0][0]=='phase_replacements'
assert outer['new_run_directories']==[ids[0][1]]
name,rid=ids[0];directory=ROOT/'data/runs'/rid;count=EXPECTED_COUNTS[name];assert count==14
rows=read_test_receipts(settings,directory,'draw-endings.json',{'draw-ending-summary.json':1,'draw-ending-methods.json':count})
summary,methods=rows[0],rows[1:]
expected=cases(build/'source/forge-gui-desktop/target/surefire-reports/testng-results.xml',set(PARTITIONS[name]))
actual=cases(directory/'rules-reports/testng-results.xml');assert actual==expected and sum(actual.values())==14
assert Counter((r['method'],tuple(r.get('parameters',[]))) for r in methods)==expected
assert all(r['status']=='PASS' for r in methods)
assert summary['counts']==dict(total=14,passed=14,failed=0,skipped=0)
assert all(summary[n]==0 for n in ('testng_status','configuration_failures','configuration_skips','partial_successes'))
assert summary['production_classes_from_packaged_jar']==list(NATIVE_CLASSES) and len(NATIVE_CLASSES)==188
assert file_hash(directory/'rules-reports/testng-results.xml')==summary['report_sha256']
assert all(file_hash(Path(n))==h for n,h in summary['inputs'].items())
execution=load(directory/'rules-process.json');assert execution['worker_sha256']==file_hash(ROOT/'tests/support/draw_ending_worker.py')
assert execution['process']['status']=='finished' and execution['process']['returncode']==0 and execution['process']['elapsed_seconds']<=90
assert all(file_hash(Path(n))==h for n,h in execution['process']['support_source_hashes'].items())
classes=load(directory/'rules-inputs.json')['classes'];assert all(file_hash(directory/'rules-classes'/n)==h for n,h in classes.items())
assert (ROOT/'config/engine_config.json').read_bytes()==(RUN/'active-config-before.json').read_bytes()
record={'build_id':build.name,'group':name,'run_id':rid,'counts':summary['counts'],'native_method_and_parameter_counter_matches':True,'production_origins':188,'worker_limit_seconds':90,'wrapper_limit_seconds':420,'configuration_restored_byte_identically':True,'report_sha256':summary['report_sha256'],'process':execution['process'],'scope':'Fourteen registered phase cases via regular packaged worker only. Not full3775 or original1271/90, full legal histories/games or universal card rules proof.'}
(RUN/'audit.json').write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8',newline='\n');print(json.dumps(record),flush=True)
