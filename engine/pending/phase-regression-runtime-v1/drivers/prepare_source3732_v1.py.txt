from collections import Counter
from pathlib import Path
import json,re,shutil,subprocess,sys
ROOT=Path.cwd();sys.path.insert(0,str(ROOT))
from commander.engine.forge_config import file_hash
from tools import build_sticker_engine as builder
WORK=ROOT/'data/development/commander-multiplayer-attack-research/waterbend-filter-research'
RUN=WORK/'source3732-v1';PRIOR=WORK/'source3718-v1';BASE=PRIOR/'source'
BUILD=ROOT/'data/engine-builds/27bb401949a2401ab440509f215df73e'
load=lambda p:json.loads(p.read_bytes())

ci=load(WORK/'phase-registry-published-ci-v1.json')
assert ci['conclusion']=='success' and ci['status']=='completed'

paths=filter(None,subprocess.check_output(['git','ls-files','-z']).decode('utf-8').split('\0'));frozen={n:file_hash(ROOT/n) for n in paths}

builder._copy_regressions(source)
assert all(file_hash(source/n)==h for n,h in before_tests.items())
assert all(file_hash(source/n)==h for n,h in production.items())

assert set(all_tests)-set(before_tests)=={fixture}

expected=Counter((v['method'],tuple(v['parameters'])) for v in prior['cases']);assert sum(expected.values())==3718
names=re.findall(r'@Test\s+public void (\w+)\(', (source/fixture).read_text(encoding='utf-8'))
assert len(names)==len(set(names))==14
added=Counter(('forge.ai.CommanderPhaseReplacementTest.'+n,()) for n in names)
assert not set(added)&set(expected);expected.update(added)

save(RUN/'selection.json',{'build':str(BUILD),'source':str(source),'suite':{'classes':[*selectors,'CommanderPhaseReplacementTest'],'expected_cases':3732,'expected_manifest_sha256':file_hash(RUN/'expected-cases.json')},'changed_sources':[],'fresh_target_directories':True,'scope':'All3718 prior source cases plus14 phase replacement controls, exact method/parameter membership. selected[fixture]=file_hash(source/fixture)
for name,value in [('test-inputs.json',selected),('production-inputs.json',production),('selected-test-inputs.json',all_tests),('card-inputs.json',cards),('root-inputs.json',frozen),('source-ci.json',ci)]:save(RUN/name,value)
save(RUN/'preparation.json',{'passed':True,'production_sources':2168,'changed_sources':0,'all_java_test_sources_frozen':542,'selected_test_inputs':182,'card_inputs':350,'expected_cases':3732,'retained_prior_cases':3718,'new_cases':14,'all_prior_test_bytes_unchanged':True})
