"""Preserve the repaired parameter-aware validation cohort without activating it."""
from pathlib import Path
import hashlib
import json
import re
import shutil

ROOT = Path(__file__).resolve().parents[4]
WORK = Path(__file__).resolve().parent
DEST = ROOT / 'engine/pending/parameter-runtime-v1'
INDEX = ROOT / 'engine/parameter_runtime_evidence.json'
load = lambda p: json.loads(p.read_bytes())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()


def main():
    assert not DEST.exists() and not INDEX.exists()
    epoch = load(WORK / 'parameter-case-epoch-v1/source-epoch.json')
    smoke = load(WORK / 'parameter-case-native22-v1/audit.json')
    front = load(WORK / 'parameter-front-packaged-v1/audit.json')
    remainder = load(WORK / 'parameter-remainder-v1/audit.json')
    other = load(WORK / 'parameter-other-packaged-v1/audit.json')
    broad = load(WORK / 'parameter-broad-v1/outcome.json')
    restored = load(WORK / 'parameter-broad-v1/wrapper-process.json')
    catalog = load(WORK / 'parameter-catalog-health-v1/outcome.json')
    assert smoke['passed'] and smoke['counts']['passed'] == 22
    assert front['total_cases'] + remainder['total_cases'] == 2481
    assert other['total_cases'] == 405
    assert broad['passed'] and restored['policies_restored']
    assert restored['production_unchanged']
    assert catalog['catalog_status'] == 'passed'
    assert catalog['default_configuration_unchanged']
    frozen = load(WORK / 'parameter-case-epoch-v1/source-frozen-inputs.json')
    assert all(sha(ROOT / n) == h for n, h in frozen.items())
    files = {}

    def preserve(source, relative):
        target = DEST / relative
        assert not target.exists()
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source, target)
        assert source.read_bytes() == target.read_bytes()
        files[target.relative_to(ROOT).as_posix()] = sha(target)

    cohorts = ['parameter-case-epoch-v1', 'parameter-case-native22-v1',
               'parameter-front-packaged-v1', 'parameter-remainder-v1',
               'parameter-other-packaged-v1', 'parameter-broad-v1',
               'parameter-catalog-health-v1']
    health = load(WORK / 'parameter-catalog-health-v1/health.json')
    run_ids = {smoke['run_id'], catalog['catalog_run_id']}
    if health.get('run_id'):
        run_ids.add(health['run_id'])
    for name in cohorts:
        directory = WORK / name
        for source in sorted(directory.iterdir()):
            if source.is_file():
                preserve(source, name + '/' + source.name)
        process = directory / 'process.json'
        if process.exists():
            run_ids.update(load(process).get('new_run_directories', []))
    for suite in ('gameplay', 'serialization'):
        directory = WORK / 'parameter-broad-v1' / suite
        for name in ('selection.json', 'process.json', 'outcome.json',
                     'native-rules.log', 'testng-results.xml'):
            preserve(directory / name, 'parameter-broad-v1/' + suite + '/' + name)
    for directory_name in ('policy-originals', 'policy-applied'):
        directory = WORK / 'parameter-broad-v1' / directory_name
        for source in sorted(directory.rglob('*.java')):
            preserve(source, 'parameter-broad-v1/' + directory_name + '/' + source.relative_to(directory).as_posix())
    for run_id in sorted(run_ids):
        assert re.fullmatch('[a-f0-9]{32}', run_id)
        directory = ROOT / 'data/runs' / run_id
        assert directory.is_dir()
        for source in sorted(directory.rglob('*')):
            if source.is_file() and source.suffix in ('.json', '.jsonl', '.xml', '.log'):
                preserve(source, 'runs/' + run_id + '/' + source.relative_to(directory).as_posix())
    scripts = ['run_parameter_case_native22_v1.py', 'run_parameter_front_packaged_v1.py',
               'audit_parameter_front_packaged_v1.py', 'run_parameter_remainder_v1.py',
               'audit_parameter_remainder_v1.py', 'run_parameter_other_packaged_v1.py',
               'audit_parameter_other_packaged_v1.py', 'prepare_parameter_broad_v1.py',
               'run_parameter_broad_v1.py', 'run_parameter_catalog_health_v1.py']
    for name in scripts:
        preserve(WORK / name, 'drivers/' + name + '.txt')
    preserve(Path(__file__), 'publisher.py.txt')
    record = {
        'status': 'parameter_aware_packaged_broader_and_catalog_checks_passed',
        'source_epoch': epoch,
        'native_build': '35e8b8bcca1b41a1acc2aaa672d12b45',
        'parameter_smoke': smoke,
        'main_packaged_nonresolution_cases': 2481,
        'additional_packaged_cases': 405,
        'broader': broad,
        'catalog': catalog,
        'original_resolution_cases': 1271,
        'original_resolution_worker_seconds': 90,
        'original_resolution_gate_passed': False,
        'runtime_activated': False,
        'scope': 'Exact case/parameter and origin audits support these finite suites. Catalog construction does not prove ability correctness. Full3752 timing, full games, universal card/rules correctness and candidate activation remain unproven. Earlier failed validator evidence is retained separately.',
        'files': files,
    }
    INDEX.write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8', newline='\n')
    assert all(sha(ROOT / n) == h for n, h in files.items())
    print(json.dumps({'preserved_files': len(files), 'run_directories': len(run_ids),
                      'index': INDEX.relative_to(ROOT).as_posix()}), flush=True)


if __name__ == '__main__':
    main()
