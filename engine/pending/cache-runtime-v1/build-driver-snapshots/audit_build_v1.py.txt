def save(path,value):
    assert not path.exists(),path
    path.write_text(json.dumps(value,indent=2)+"\n",encoding="utf-8",newline="\n")
def methods(path, require_pass=True):
    rows=Counter()
    for cls in ET.parse(path).findall(".//class"):
        for method in cls.findall("test-method"):
            if method.get("is-config")!="true":
                if require_pass: assert method.get("status")=="PASS"
                rows[cls.get("name")+"."+method.get("name")]+=1
    return rows

endings=[]

with zipfile.ZipFile(selected_settings.home/selected["jar_name"]) as first,zipfile.ZipFile(settings.home/candidate["jar_name"]) as second:
    a={n:hashlib.sha256(first.read(n)).hexdigest() for n in first.namelist() if n.endswith(".class")}
    b={n:hashlib.sha256(second.read(n)).hexdigest() for n in second.namelist() if n.endswith(".class")}
    changed=sorted(n for n in a.keys()&b.keys() if a[n]!=b[n]);added=sorted(b.keys()-a.keys());removed=sorted(a.keys()-b.keys())
    source_files=load(ROOT/"engine/target_tax_evidence.json")["production_files"]+["forge-ai/src/main/java/forge/ai/simulation/MultiplayerAttackPlanner.java","forge-game/src/main/java/forge/game/card/CardState.java"]

    assert len(families)==10
    belongs=lambda name:any(name==family+".class" or name.startswith(family+"$") for family in families)
    assert added==["forge/ai/ComputerUtilMana$FilterActivationResult.class"],added
    assert removed==["forge/ai/ComputerUtilCost$1.class"],removed
    assert all(belongs(name) for name in changed+added+removed),(changed,added,removed)
    assert {family+".class" for family in families}<=set(changed),changed
