"""Run the existing 616 contracts with the native baseline or event prefilter."""
from collections import Counter
from pathlib import Path
import ast
import importlib
import json
import subprocess
import sys
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT))
from commander.engine.engine_process import ProcessLimits, run_bounded
from commander.engine.forge_config import read_settings, file_hash, jvm_library
from commander.engine.native_runtime import _engine_profile, _environment, require_bridge
from tests.support import packaged_rules, draw_ending_worker

WORK = Path(__file__).resolve().parent
MODE = sys.argv[1]
assert MODE in ('baseline', 'prototype')
assert sys.argv[2:] in ([], ['--worker'])
RUN = WORK / ('event-prefilter616-' + MODE + '-v1')
ORIGINAL = ROOT / 'data/runs/dd055d84f753460cbb78c4bce32c5a73'
PROTOTYPE = WORK / 'replacement-event-prefilter-prototype-v1'
EPOCH = WORK / 'remaining-packaged-epoch-v1'
SETTINGS = ROOT / 'data/engine-builds/1bae22d83a0743fc909573302ed9e3c0/engine-config.json'
load = lambda path: json.loads(path.read_bytes())
parsed = ast.parse((WORK / 'run_keyword_membership616_v3.py').read_text(encoding='utf-8'))
CLASSES = next(ast.literal_eval(item.value) for item in parsed.body if isinstance(item, ast.Assign)
               and any(isinstance(target, ast.Name) and target.id == 'CLASSES' for target in item.targets))
CLASSES += ('forge.ai.CommanderPrintingSelectionTest', 'forge.ai.CommanderKeywordMembershipTest')
assert len(CLASSES) == len(set(CLASSES))


def save(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n', encoding='utf-8', newline='\n')


def cases(path, classes=None):
    rows = Counter()
    for cls in ET.parse(path).findall('.//class'):
        if classes is not None and cls.get('name') not in classes:
            continue
        for method in cls.findall('test-method'):
            if method.get('is-config') == 'true':
                assert method.get('status') == 'PASS'
                continue
            assert method.get('status') == 'PASS'
            params = tuple((v.text or '').strip() for v in method.findall('./params/param/value'))
            rows[(cls.get('name') + '.' + method.get('name'), params)] += 1
    return rows


def verify():
    metadata = load(ORIGINAL / 'rules-inputs.json')
    assert metadata['classes'] == {p.relative_to(ORIGINAL / 'rules-classes').as_posix(): file_hash(p)
                                   for p in (ORIGINAL / 'rules-classes').rglob('*.class')}
    assert all(file_hash(Path(n)) == h for n, h in metadata['inputs'].items())
    inputs = load(PROTOTYPE / 'inputs.json')
    assert all(file_hash(PROTOTYPE / n) == h for n, h in inputs.items())
    assert all(file_hash(ROOT / n) == h for n, h in load(EPOCH / 'source-frozen-inputs.json').items())
    return inputs


if '--worker' in sys.argv:
    inputs = verify()
    settings = read_settings(SETTINGS)
    engine = settings.verify()
    require_bridge()
    java = importlib.import_module('jpype')
    recipe = load(ROOT / 'engine/sticker_recipe.json')
    dependency = ROOT / recipe['packaged_testng']['path']
    archive = settings.home / engine['jar_name']
    assert file_hash(dependency) == recipe['packaged_testng']['sha256']
    classpath = (() if MODE == 'baseline' else (PROTOTYPE / 'classes',)) + (archive, ORIGINAL / 'rules-classes', dependency)
    java.startJVM(jvm_library(settings.java), '-Xmx2048m', '-XX:ActiveProcessorCount=2',
                  '-Djava.awt.headless=true', '-Dfile.encoding=UTF-8',
                  classpath=list(map(str, classpath)), convertStrings=True)
    runtime = java.JClass('java.lang.Runtime').getRuntime()
    assert runtime.availableProcessors() == 2 and runtime.maxMemory() <= 2147483648
    overrides = set() if MODE == 'baseline' else {n[len('classes/'):] for n in inputs if n.startswith('classes/')}
    for name in draw_ending_worker.NATIVE_CLASSES:
        expected = PROTOTYPE / 'classes' if name.replace('.', '/') + '.class' in overrides else archive
        assert packaged_rules.class_origin(java, name).samefile(expected), name
    for name in overrides:
        assert packaged_rules.class_origin(java, name[:-6].replace('/', '.')).samefile(PROTOTYPE / 'classes')
    for name in (*CLASSES, 'forge.ai.AITest', 'forge.ai.CommanderPrintingSelectionExpected'):
        assert packaged_rules.class_origin(java, name).samefile(ORIGINAL / 'rules-classes')
    runner = java.JClass('org.testng.TestNG')()
    listener = java.JClass('org.testng.TestListenerAdapter')()
    runner.addListener(listener)
    runner.setTestClasses(java.JArray(java.JClass('java.lang.Class'))([java.JClass(n).class_ for n in CLASSES]))
    runner.setOutputDirectory(str(RUN / 'reports'))
    runner.setUseDefaultListeners(True)
    runner.run()
    counts, methods = packaged_rules.test_results(listener)
    expected = cases(SETTINGS.parent / 'source/forge-gui-desktop/target/surefire-reports/testng-results.xml', set(CLASSES))
    actual = cases(RUN / 'reports/testng-results.xml')
    assert actual == expected and sum(actual.values()) == 616
    outcome = {'mode': MODE, 'counts': counts, 'methods': methods, 'native_method_and_parameter_counter_matches': True,
               'configuration_failures': len(listener.getConfigurationFailures()), 'configuration_skips': len(listener.getConfigurationSkips()),
               'testng_status': int(runner.getStatus()), 'actual_processors': int(runtime.availableProcessors()),
               'max_java_heap_bytes': int(runtime.maxMemory()), 'production_origins': len(draw_ending_worker.NATIVE_CLASSES),
               'scope': 'Existing 616 safety contracts and 648 printing vectors unchanged; no independent packaged resolution pass or full rules proof.'}
    save(RUN / 'outcome.json', outcome)
    assert counts == dict(total=616, passed=616, failed=0, skipped=0)
    assert outcome['configuration_failures'] == outcome['configuration_skips'] == outcome['testng_status'] == 0
    verify()
    java.shutdownJVM()
else:
    verify()
    assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=ROOT)
    ci = load(EPOCH / 'source-ci.json')
    assert ci['head_sha'] == subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
    assert ci['conclusion'] == 'success' and ci['status'] == 'completed'
    RUN.mkdir()
    (RUN / 'driver.py.txt').write_bytes(Path(__file__).read_bytes())
    save(RUN / 'selection.json', {'classes': CLASSES, 'expected_cases': 616, 'mode': MODE})
    settings = read_settings(SETTINGS)
    with _engine_profile(settings, RUN):
        process = run_bounded([sys.executable, '-I', str(Path(__file__)), MODE, '--worker'], settings.home,
                              RUN / 'native.log', ProcessLimits(90, log_bytes=65536, environment=_environment(RUN)))
    save(RUN / 'process.json', process)
    verify()
    assert file_hash(Path(__file__)) == file_hash(RUN / 'driver.py.txt')
    outcome = load(RUN / 'outcome.json') if (RUN / 'outcome.json').exists() else {}
    print(json.dumps({'process': process, 'counts': outcome.get('counts'), 'mode': MODE}), flush=True)
    assert process['status'] == 'finished' and process['returncode'] == 0
