results={};grouped_methods=Counter()

    run=ROOT/'data/runs'/run_id
    rows=read_test_receipts(settings,run,manifest,{prefix+'-summary.json':1,prefix+'-methods.json':count})
    summary,outcomes=rows[0],rows[1:]
    actual=Counter(row['method'] for row in outcomes)
    expected=Counter({method:n for method,n in methods.items() if method.rsplit('.',1)[0] in classes})
    assert sum(expected.values())==count and actual==expected
    assert all(row['status']=='PASS' for row in outcomes)
    assert summary['counts']==dict(total=count,passed=count,failed=0,skipped=0)
    assert summary['production_classes_from_packaged_jar']==list(origins) and len(set(origins))==len(origins)
    assert all(summary[key]==0 for key in ('testng_status','configuration_failures','configuration_skips','partial_successes'))
    assert file_hash(run/'rules-reports/testng-results.xml')==summary['report_sha256']
    assert all(file_hash(Path(path))==sha for path,sha in summary['inputs'].items())
    execution=load(run/'rules-process.json')
    assert execution['worker_sha256']==file_hash(Path(worker.__file__))
    assert execution['process']['status']=='finished' and execution['process']['returncode']==0 and execution['process']['elapsed_seconds']<=90
    assert all(file_hash(Path(path))==sha for path,sha in execution['process']['support_source_hashes'].items())
    artifacts=[bound(run/item) for item in (manifest,prefix+'-summary.json',prefix+'-methods.json','rules-inputs.json','rules-process.json','rules-reports/testng-results.xml')]
    for chunk in load(run/(prefix+'-methods.json')).get('chunks',[]):
        assert file_hash(run/chunk['name'])==chunk['sha256'];artifacts.append(bound(run/chunk['name']))
    results[name]={'run':run_id,'counts':summary['counts'],'production_origins':len(origins),'process':execution['process'],'artifacts':artifacts}
    if name in PARTITIONS:
        grouped_methods.update(actual)
