from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import sys
ROOT=Path.cwd();sys.path.insert(0,str(ROOT))
from commander.engine.forge_config import read_settings,file_hash

ci={'id':45655802433,'head_sha':head,'status':'completed','conclusion':'success'}
save(EPOCH/'source-ci.json',ci)
tracked=[n for n in subprocess.check_output(['git','ls-files','-z'],cwd=ROOT).decode('utf-8').split('\0') if n]
frozen={n:file_hash(ROOT/n) for n in tracked}
save(EPOCH/'source-frozen-inputs.json',frozen);save(EPOCH/'source-epoch.json',{'head':head,'tracked_files':len(frozen),'ci_id':ci['id']})

    for dest,origin in [(PROTOTYPE/'source'/n,p),(PROTOTYPE/'baseline'/n,canonical/n)]:
        dest.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(origin,dest)
