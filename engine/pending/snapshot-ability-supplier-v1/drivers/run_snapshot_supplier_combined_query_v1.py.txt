"""Run isolated EnumSet query contracts or fixed escaping-allocation controls."""
from collections import Counter
from pathlib import Path
import importlib
import json
import re
import subprocess
import sys
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT))
from commander.engine.engine_process import ProcessLimits, run_bounded
from commander.engine.forge_config import file_hash, jvm_library, read_settings
from commander.engine.native_runtime import _engine_profile, _environment, require_bridge
from tests.support import draw_ending_worker, packaged_rules

WORK = Path(__file__).resolve().parent
PROTOTYPE = WORK / "snapshot-supplier-combined-prototype-v1"
FIXTURE = WORK / "enum-card-scan-capacity-fixture-v1"
EPOCH = WORK / "post-sacrifice-runtime-epoch-v1"
load = lambda path: json.loads(path.read_bytes())
MODE, FAMILY = sys.argv[1:3]
assert MODE in ("baseline", "prototype")
assert FAMILY in ("contracts", "allocation")
assert sys.argv[3:] in ([], ["--worker"])
WORKER = sys.argv[3:] == ["--worker"]
RUN = WORK / ("snapshot-supplier-combined-query-" + FAMILY + "-" + MODE + "-v1")
CONFIG = ROOT / load(WORK / "sacrifice-candidate-v1/candidate-selection.json")["build"] / "engine-config.json"
UPSTREAM = CONFIG.parent / "source/forge-gui-desktop/target/test-classes"
SIMPLE = "CommanderEnumCardScanCapacityTest" if FAMILY == "contracts" else "CommanderEnumCardScanAllocationTest"
TARGET = "forge.ai." + SIMPLE
COUNT = 17 if FAMILY == "contracts" else 1


def save(path, value):
    path.write_text(json.dumps(value, indent=2) + "\n", encoding="utf-8", newline="\n")


def verify(root=False):
    for directory in (PROTOTYPE, FIXTURE):
        assert all(file_hash(directory / name) == digest for name, digest in load(directory / "inputs.json").items())
    changed = load(PROTOTYPE / "proposal.json")["changed_sources"]
    manifest = load(WORK / "source3622-v1/production-inputs.json")
    assert len(changed) == len(set(changed)) == 17
    assert all(file_hash(PROTOTYPE / "baseline" / name) == manifest[name] for name in changed)
    copier = (PROTOTYPE / "source/forge-ai/src/main/java/forge/ai/simulation/GameCopier.java").read_text(encoding="utf-8")
    assert all(name in copier for name in ("copySnapshotGameSpellCountFrom", "copySnapshotDiscardedFrom", "copySnapshotSacrificedFrom"))
    assert not (PROTOTYPE / "classes/forge/game/player/Player.class").exists()
    assert load(RUN / "test-inputs.json")["files"] == {
        path.relative_to(UPSTREAM).as_posix(): file_hash(path)
        for path in UPSTREAM.rglob("*") if path.is_file()
    }
    if root:
        assert all(file_hash(ROOT / name) == digest for name, digest in load(EPOCH / "source-frozen-inputs.json").items())


def cases(report):
    result = Counter()
    for cls in ET.parse(report).findall(".//class"):
        for method in cls.findall("test-method"):
            if method.get("is-config") != "true":
                parameters = tuple((value.text or "").strip() for value in method.findall("./params/param/value"))
                result[(cls.get("name") + "." + method.get("name"), parameters)] += 1
    return result


if WORKER:
    verify()
    settings = read_settings(CONFIG)
    lock = settings.verify()
    assert lock["native_tests"]["total"] == 4822
    require_bridge()
    java = importlib.import_module("jpype")
    recipe = load(ROOT / "engine/sticker_recipe.json")
    dependency = ROOT / recipe["packaged_testng"]["path"]
    assert file_hash(dependency) == recipe["packaged_testng"]["sha256"]
    archive = settings.home / lock["jar_name"]
    paths = (() if MODE == "baseline" else (PROTOTYPE / "classes",)) + (archive, FIXTURE / "classes", UPSTREAM, dependency)
    java.startJVM(jvm_library(settings.java), "-Xmx2048m", "-XX:ActiveProcessorCount=2",
                  "-Djava.awt.headless=true", "-Dfile.encoding=UTF-8", classpath=list(map(str, paths)), convertStrings=True)
    runtime = java.JClass("java.lang.Runtime").getRuntime()
    assert runtime.availableProcessors() == 2 and runtime.maxMemory() == 2147483648
    overrides = set() if MODE == "baseline" else {
        name[len("classes/"):] for name in load(PROTOTYPE / "inputs.json")
        if name.startswith("classes/") and name.endswith(".class")
    }
    families = [name.split("/src/main/java/")[1][:-5] for name in load(PROTOTYPE / "proposal.json")["changed_sources"]]
    assert all(any(name == family + ".class" or name.startswith(family + "$") for family in families) for name in overrides)
    for name in tuple(dict.fromkeys((*draw_ending_worker.NATIVE_CLASSES, "forge.game.Game"))):
        expected_origin = PROTOTYPE / "classes" if name.replace(".", "/") + ".class" in overrides else archive
        assert packaged_rules.class_origin(java, name).samefile(expected_origin), name
    for name in overrides:
        assert packaged_rules.class_origin(java, name[:-6].replace("/", ".")).samefile(PROTOTYPE / "classes"), name
    assert packaged_rules.class_origin(java, TARGET).samefile(FIXTURE / "classes")
    assert packaged_rules.class_origin(java, "forge.ai.AITest").samefile(UPSTREAM)
    names = re.findall(r"@Test\s+public void (\w+)\(", (FIXTURE / (SIMPLE + ".java")).read_text(encoding="utf-8"))
    assert len(names) == len(set(names)) == COUNT
    expected = Counter((TARGET + "." + name, ()) for name in names)
    runner = java.JClass("org.testng.TestNG")()
    listener = java.JClass("org.testng.TestListenerAdapter")()
    runner.addListener(listener)
    runner.setTestClasses(java.JArray(java.JClass("java.lang.Class"))([java.JClass(TARGET).class_]))
    runner.setOutputDirectory(str(RUN / "reports"))
    runner.setUseDefaultListeners(True)
    runner.run()
    counts, methods = packaged_rules.test_results(listener)
    actual = cases(RUN / "reports/testng-results.xml")
    allocation = {}
    if FAMILY == "allocation":
        allocation = {str(entry.getKey()): int(entry.getValue()) for entry in java.JClass(TARGET).allocationBytes().entrySet()}
    outcome = {
        "mode": MODE, "family": FAMILY, "counts": counts, "methods": methods,
        "exact_case_counter_matches": actual == expected,
        "configuration_failures": len(listener.getConfigurationFailures()),
        "configuration_skips": len(listener.getConfigurationSkips()),
        "testng_status": int(runner.getStatus()), "allocated_bytes_by_card_count": allocation,
        "actual_processors": int(runtime.availableProcessors()), "actual_max_heap_bytes": int(runtime.maxMemory()),
        "all_overlay_origins_verified": True, "warmup_queries": 1000, "measured_queries": 2000,
        "scope": "Unpromoted17-source snapshot/view/cache plus bounded allocation-hint combination. Supplied-world query contracts or fixed escaping-allocation counts; no elapsed-speedup, retained-memory, original1271/90 or full-rules proof."
    }
    save(RUN / "outcome.json", outcome)
    assert counts == dict(total=COUNT, passed=COUNT, failed=0, skipped=0) and actual == expected
    assert outcome["configuration_failures"] == outcome["configuration_skips"] == outcome["testng_status"] == 0
    if FAMILY == "allocation":
        assert set(allocation) == {"0", "8", "80", "400"} and all(value >= 0 for value in allocation.values())
    verify()
    java.shutdownJVM()
else:
    assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=ROOT)
    ci = load(EPOCH / "source-ci.json")
    assert ci["head_sha"] == subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    assert ci["status"] == "completed" and ci["conclusion"] == "success"
    RUN.mkdir()
    save(RUN / "test-inputs.json", {"files": {
        path.relative_to(UPSTREAM).as_posix(): file_hash(path)
        for path in UPSTREAM.rglob("*") if path.is_file()
    }})
    verify(root=True)
    (RUN / "driver.py.txt").write_bytes(Path(__file__).read_bytes())
    save(RUN / "selection.json", {"mode": MODE, "family": FAMILY, "class": TARGET,
                                    "expected_cases": COUNT, "worker_limit_seconds": 90})
    settings = read_settings(CONFIG)
    with _engine_profile(settings, RUN):
        process = run_bounded([sys.executable, "-I", str(Path(__file__)), MODE, FAMILY, "--worker"],
                              settings.home, RUN / "native.log", ProcessLimits(90, log_bytes=65536, environment=_environment(RUN)))
    save(RUN / "process.json", process)
    verify(root=True)
    assert file_hash(Path(__file__)) == file_hash(RUN / "driver.py.txt")
    outcome = load(RUN / "outcome.json") if (RUN / "outcome.json").exists() else {}
    print(json.dumps({"mode": MODE, "family": FAMILY, "process": process, "counts": outcome.get("counts"),
                      "allocated_bytes_by_card_count": outcome.get("allocated_bytes_by_card_count")}), flush=True)
    assert process["status"] == "finished" and process["returncode"] == 0, process
