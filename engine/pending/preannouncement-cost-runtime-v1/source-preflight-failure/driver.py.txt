"""Freeze the reviewed commit and validate every retained source case plus nine boundaries."""
from collections import Counter
from pathlib import Path
from types import SimpleNamespace
import hashlib
import json
import shutil
import subprocess
import sys
import time
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT))
from tools import build_sticker_engine as builder
WORK = Path(__file__).resolve().parent
SOURCE = ROOT / 'data/development/commander-blocking-history-research/source'
PRIOR = WORK / 'source3511-v1'
EPOCH = WORK / 'preannouncement-source-epoch-v1'
RUN = WORK / 'source3520-v1'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
load = lambda p: json.loads(p.read_bytes())
def save(p, value):
    p.write_text(json.dumps(value, indent=2)+'\n',encoding='utf-8',newline='\n')
def cases(report):
    tree = ET.parse(report).getroot()
    methods = []
    for cls in tree.findall('.//class'):
        for item in cls.findall('test-method'):
            assert item.get('status') == 'PASS', (cls.get('name'), item.attrib)
            if item.get('is-config') != 'true':
                methods.append({'method':cls.get('name')+'.'+item.get('name'),
                    'parameters':[(v.text or '').strip() for v in item.findall('./params/param/value')],
                    'status':item.get('status')})
    return {n:int(tree.get(n)) for n in ('total','passed','failed','skipped')}, methods

assert not subprocess.check_output(['git','status','--porcelain'],cwd=ROOT)
head = subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()
ci = load(EPOCH/'source-ci.json')
assert ci['head_sha']==head and ci['conclusion']=='success' and ci['status']=='completed'
assert not (EPOCH/'source-frozen-inputs.json').exists() and not RUN.exists()
tracked = subprocess.check_output(['git','ls-files','-z'],cwd=ROOT).decode('utf-8').split('\0')
frozen = {n:sha(ROOT/n) for n in tracked if n}
save(EPOCH/'source-frozen-inputs.json', frozen)
save(EPOCH/'source-epoch.json', {'head':head,'tracked_files':len(frozen),'ci_id':ci['id']})
oldproduction = load(PRIOR/'production-inputs.json')
assert len(oldproduction)==2168 and all(sha(SOURCE/n)==h for n,h in oldproduction.items())

assert not (SOURCE/fixture).exists()

assert production.keys()==oldproduction.keys()
changed = [n for n in production if production[n]!=oldproduction[n]]
assert changed==['forge-ai/src/main/java/forge/ai/ComputerUtilCost.java']
assert production[changed[0]]==load(WORK/'preannouncement-broad-v1/proposed-production.json')['proposed_sha256']
