from collections import Counter
from pathlib import Path
import json,shutil,subprocess,sys

assert all(file_hash(source/n)==file_hash(BASE/n) for n in production)
before_tests={p.relative_to(source).as_posix():file_hash(p) for p in source.glob('*/src/test/java/**/*.java')}

for name,options in [('check',['--check']),('apply',[])]:
 command=[git,'apply',*options,'--directory='+source.relative_to(ROOT).as_posix(),str(patch)]
 result=run_bounded(command,ROOT,RUN/('patch-'+name+'.log'),ProcessLimits(90,environment=_environment(RUN)));save(RUN/('patch-'+name+'-process.json'),result)
 assert result['status']=='finished' and result['returncode']==0,result
assert [n for n in production if normalize(source/n)!=normalize(BASE/n)]==changed
assert all(normalize(source/n)==normalize(PROTO/'source'/n) for n in changed)
builder._copy_regressions(source)
assert all(file_hash(source/n)==h for n,h in before_tests.items())
all_tests={p.relative_to(source).as_posix():file_hash(p) for p in source.glob('*/src/test/java/**/*.java')};assert len(all_tests)==538
fixture='forge-gui-desktop/src/test/java/forge/ai/CommanderCopierReleaseTest.java'
assert all_tests[fixture]==recipe['commander_copier_release_test_sha256']

expected=Counter((r['method'],tuple((v or '').strip() for v in r['parameters'])) for r in prior['methods']);assert sum(expected.values())==3679
control=WORK/'copier-release-native4879-release-prototype-v3';assert load(control/'outcome.json')['counts']==dict(total=8,passed=8,failed=0,skipped=0)

for cls in ET.parse(control/'reports/testng-results.xml').findall('.//class'):

  if m.get('is-config')!='true':
   assert cls.get('name')=='forge.ai.CommanderCopierReleaseTest' and m.get('status')=='PASS'
   added[(cls.get('name')+'.'+m.get('name'),tuple((v.text or '').strip() for v in m.findall('./params/param/value')))]+=1
assert sum(added.values())==8 and not (set(added)&set(expected));expected.update(added)
save(RUN/'expected-cases.json',[{'method':m,'parameters':list(p),'multiplicity':n} for (m,p),n in sorted(expected.items())])
original=load(WORK/'source3679-v2/selection.json')
selectors=next(a.removeprefix('-Dtest=') for a in original['command'] if a.startswith('-Dtest=')).split(',')

Not original1271/90, a sealed native build, runtime activation or complete rules proof.'})
save(RUN/'production-inputs.json',{n:file_hash(source/n) for n in production});save(RUN/'selected-test-inputs.json',all_tests)
selected=load(WORK/'source3679-v2/test-inputs.json');assert len(selected)==177 and all(file_hash(source/n)==file_hash(BASE/n) for n in selected)

cards=load(WORK/'source3679-v2/card-inputs.json');assert len(cards)==350 and all(file_hash(source/n)==h for n,h in cards.items());save(RUN/'card-inputs.json',cards)
save(RUN/'root-inputs.json',frozen);save(RUN/'source-ci.json',ci)
save(RUN/'preparation.json',{'passed':True,'production_sources':2168,'changed_sources':1,'all_java_test_sources_frozen':538,'retained_selected_test_inputs':178,'card_inputs':350,'expected_cases':3687,'all_prior_cases_preserved':3679,'new_cases':8,'registered_patch_applied_and_matches_tested_prototype':True})
print('Clean source prepared:2168 production,538 frozen Java tests,178 selected inputs,350 card inputs,all3687 cases.',flush=True)
